cmake_minimum_required(VERSION 3.24)
project(FootageRecordCaptureCheck LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The Windows capture check requires the Windows SDK and MSVC.")
endif()
add_executable(FootageRecordCaptureCheck WIN32 src/main.cpp app.manifest)
target_compile_features(FootageRecordCaptureCheck PRIVATE cxx_std_20)
target_compile_definitions(FootageRecordCaptureCheck PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(FootageRecordCaptureCheck PRIVATE /W4 /permissive- /EHsc)
target_link_libraries(FootageRecordCaptureCheck PRIVATE d3d11 dxgi windowsapp comctl32)
# Avoid a separate VC runtime installer for this self-contained capture check.
set_property(TARGET FootageRecordCaptureCheck PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
install(TARGETS FootageRecordCaptureCheck RUNTIME DESTINATION .)
